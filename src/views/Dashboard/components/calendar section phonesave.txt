import { useState, useEffect, memo, useMemo, useRef } from 'react';
import { ChevronLeft, ChevronRight, Save, Loader2, Ban, CalendarCheck, X, Calendar as CalendarIcon, Info, CalendarDays } from 'lucide-react';
import { motion, AnimatePresence, animate } from 'framer-motion';
import { authService } from '../../../services/authService';

// --- LINIOWY LICZNIK (Szybki i stały postęp) ---
const RollingNumber = ({ value }: { value: number }) => {
    const [display, setDisplay] = useState(0);
    const prevValue = useRef(0);

    useEffect(() => {
        const controls = animate(prevValue.current, value, {
            duration: 0.4, // Szybki czas trwania
            ease: "linear", // Stała prędkość, bez zwalniania na końcu
            onUpdate: (latest) => setDisplay(Math.floor(latest)),
        });
        prevValue.current = value;
        return () => controls.stop();
    }, [value]);

    return <span className="tabular-nums">{display}</span>;
};

export const CalendarSection = memo(({ addToast }: { addToast?: (msg: string, type?: any) => void }) => {
    const [isComponentReady, setIsComponentReady] = useState(false);
    const [isMobileFullscreen, setIsMobileFullscreen] = useState(false);
    const [isAnimating, setIsAnimating] = useState(false);

    useEffect(() => {
        const timer = setTimeout(() => setIsComponentReady(true), 50);
        return () => clearTimeout(timer);
    }, []);

    const [currentDate, setCurrentDate] = useState(new Date());
    const [busyDays, setBusyDays] = useState<string[]>([]);
    const [serverBusyDays, setServerBusyDays] = useState<string[]>([]);
    const [isLoading, setIsLoading] = useState(false);
    const [isSaving, setIsSaving] = useState(false);

    useEffect(() => {
        let isMounted = true;
        const fetchData = async () => {
            setIsLoading(true);
            try {
                const year = currentDate.getFullYear();
                const month = currentDate.getMonth() + 1;
                const data = await authService.getAvailability(year, month);
                if (isMounted && data?.busyDays) {
                    setBusyDays(data.busyDays);
                    setServerBusyDays(data.busyDays);
                }
            } catch (error) {
                if (addToast) addToast("Błąd grafiku", "error");
            } finally {
                if (isMounted) setIsLoading(false);
            }
        };
        if (isComponentReady) fetchData();
        return () => { isMounted = false; };
    }, [currentDate.getFullYear(), currentDate.getMonth(), isComponentReady]);

    const calendarData = useMemo(() => {
        const y = currentDate.getFullYear();
        const m = currentDate.getMonth();
        const dInM = new Date(y, m + 1, 0).getDate();
        const fDay = new Date(y, m, 1).getDay();
        const nameOnly = currentDate.toLocaleString('pl-PL', { month: 'long' });
        return {
            year: y, month: m, daysInMonth: dInM,
            firstDay: fDay === 0 ? 6 : fDay - 1,
            monthNameOnly: nameOnly.charAt(0).toUpperCase() + nameOnly.slice(1),
        };
    }, [currentDate]);

    const workingDaysCount = useMemo(() => calendarData.daysInMonth - busyDays.length, [calendarData.daysInMonth, busyDays]);

    const formatDate = (day: number) => {
        const mStr = String(calendarData.month + 1).padStart(2, '0');
        const dStr = String(day).padStart(2, '0');
        return `${calendarData.year}-${mStr}-${dStr}`;
    };

    const handleSave = async () => {
        setIsSaving(true);
        try {
            const daysOff = busyDays.filter(day => !serverBusyDays.includes(day));
            const daysOn = serverBusyDays.filter(day => !busyDays.includes(day));
            await authService.updateAvailability(daysOff, daysOn);
            setServerBusyDays(busyDays);
            if (addToast) addToast("Zapisano grafik", "success");
            setIsMobileFullscreen(false);
        } catch (error) {
            if (addToast) addToast("Błąd zapisu", "error");
        } finally { setIsSaving(false); }
    };

    const toggleDayStatus = (day: number) => {
        const dateStr = formatDate(day);
        setBusyDays(prev => prev.includes(dateStr) ? prev.filter(d => d !== dateStr) : [...prev, dateStr]);
    };

    const toggleWeekends = () => {
        const weekends = [];
        for (let d = 1; d <= calendarData.daysInMonth; d++) {
            const date = new Date(calendarData.year, calendarData.month, d);
            if (date.getDay() === 0 || date.getDay() === 6) weekends.push(formatDate(d));
        }
        const areAllBlocked = weekends.every(date => busyDays.includes(date));
        setBusyDays(prev => areAllBlocked ? prev.filter(date => !weekends.includes(date)) : Array.from(new Set([...prev, ...weekends])));
    };

    const weekendsBlocked = useMemo(() => {
        const weekends = [];
        for (let d = 1; d <= calendarData.daysInMonth; d++) {
            const date = new Date(calendarData.year, calendarData.month, d);
            if (date.getDay() === 0 || date.getDay() === 6) weekends.push(formatDate(d));
        }
        return weekends.length > 0 && weekends.every(date => busyDays.includes(date));
    }, [busyDays, calendarData]);

    const RenderCalendarGrid = ({ isMobile = false }: { isMobile?: boolean }) => (
        <div className={`relative ${isMobile ? 'flex-1 px-4' : 'bg-white rounded-[2.5rem] p-8 shadow-sm border border-gray-50'}`}>
            <div className={`flex items-center justify-between ${isMobile ? 'my-6' : 'mb-8'}`}>
                <button onClick={() => setCurrentDate(new Date(calendarData.year, calendarData.month - 1))} className="p-3 bg-gray-50 rounded-2xl text-gray-400 active:scale-90"><ChevronLeft size={isMobile ? 24 : 20} /></button>
                <div className="text-center">
                    <div className={`${isMobile ? 'text-2xl' : 'text-xl'} font-black text-gray-900 tracking-tight`}>{calendarData.monthNameOnly}</div>
                    <div className="text-[10px] font-bold text-indigo-500 uppercase tracking-widest">{calendarData.year}</div>
                </div>
                <button onClick={() => setCurrentDate(new Date(calendarData.year, calendarData.month + 1))} className="p-3 bg-gray-50 rounded-2xl text-gray-400 active:scale-90"><ChevronRight size={isMobile ? 24 : 20} /></button>
            </div>

            <div className="grid grid-cols-7 gap-2 mb-4">
                {['Pn', 'Wt', 'Śr', 'Cz', 'Pt', 'So', 'Nd'].map((d, i) => (
                    <div key={d} className={`text-[10px] font-black uppercase text-center ${i >= 5 ? 'text-rose-400' : 'text-gray-300'}`}>{d}</div>
                ))}
            </div>

            <div className="grid grid-cols-7 gap-2 md:gap-3">
                {Array.from({ length: calendarData.firstDay }).map((_, i) => <div key={`empty-${i}`} className="aspect-square" />)}
                {Array.from({ length: calendarData.daysInMonth }).map((_, i) => {
                    const day = i + 1;
                    const dateStr = formatDate(day);
                    const isBusy = busyDays.includes(dateStr);
                    const isToday = new Date().toDateString() === new Date(calendarData.year, calendarData.month, day).toDateString();

                    return (
                        <motion.button
                            key={dateStr}
                            whileTap={(!isMobile || !isAnimating) ? { scale: 0.9 } : {}}
                            onClick={() => toggleDayStatus(day)}
                            className={`aspect-square rounded-[1.25rem] flex flex-col items-center justify-center transition-all border-2 relative ${isBusy ? 'bg-rose-500 border-rose-500 text-white shadow-lg shadow-rose-100' : 'bg-white border-gray-100 text-gray-700'} ${isToday && !isBusy ? 'ring-2 ring-indigo-500' : ''}`}
                        >
                            <span className={`${isMobile ? 'text-lg' : 'text-sm'} font-black`}>{day}</span>
                            <div className={`h-1.5 w-1.5 rounded-full mt-1 ${isBusy ? 'bg-white/50' : 'bg-emerald-400'}`} />
                        </motion.button>
                    );
                })}
            </div>
        </div>
    );

    if (!isComponentReady) return <div className="w-full h-[500px]" />;

    return (
        <>
            {/* --- DESKTOP VIEW --- */}
            <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="space-y-6 text-left max-w-5xl hidden md:block px-4 md:px-0">
                <div className="flex flex-row items-end justify-between gap-4">
                    <div>
                        <h3 className="text-2xl font-bold text-gray-900 leading-tight mb-1">Twój Grafik</h3>
                        <p className="text-gray-500 font-medium text-sm italic">Zaznacz dni, w których <span className="text-rose-500 font-bold">nie pracujesz</span>.</p>
                    </div>
                    <div className="flex gap-2">
                        <button onClick={toggleWeekends} className={`px-4 py-2.5 rounded-xl text-xs font-bold transition-all border ${weekendsBlocked ? 'bg-rose-50 text-rose-600 border-rose-100' : 'bg-indigo-50 text-indigo-600 border-indigo-100'}`}>
                            {weekendsBlocked ? <CalendarCheck size={14} /> : <Ban size={14} />} Weekendy
                        </button>
                        <button onClick={handleSave} disabled={isSaving} className="bg-indigo-600 text-white px-6 py-2.5 rounded-xl font-bold text-xs shadow-lg active:scale-95 transition-all flex items-center gap-2">
                            {isSaving ? <Loader2 size={14} className="animate-spin" /> : <Save size={14} />} Zapisz
                        </button>
                    </div>
                </div>
                <RenderCalendarGrid isMobile={false} />
            </motion.div>

            {/* --- MOBILE PREVIEW CARD --- */}
            <div className="md:hidden px-2">
                <motion.div
                    initial={{ opacity: 0, scale: 0.95 }}
                    animate={{ opacity: 1, scale: 1 }}
                    className="bg-white rounded-[2.5rem] p-8 border border-gray-100 shadow-xl shadow-indigo-50/50 relative overflow-hidden"
                >
                    <div className="absolute top-0 right-0 p-6 opacity-10"><CalendarIcon size={120} /></div>
                    <div className="relative z-10">
                        <div className="flex items-center gap-3 mb-6">
                            <div className="w-12 h-12 bg-indigo-600 rounded-2xl flex items-center justify-center text-white shadow-lg shadow-indigo-200">
                                <CalendarIcon size={24} />
                            </div>
                            <div>
                                <h4 className="font-black text-gray-900 text-xl tracking-tight">Dostępność</h4>
                                <p className="text-[10px] uppercase font-bold text-indigo-500 tracking-widest">{calendarData.monthNameOnly} {calendarData.year}</p>
                            </div>
                        </div>

                        <div className="grid grid-cols-2 gap-3 mb-8">
                            <div className="bg-gray-50 p-4 rounded-3xl min-w-0">
                                <p className="text-[10px] font-bold text-gray-400 uppercase mb-1">Pracujące</p>
                                <div className="text-sm font-black text-emerald-500 flex items-center gap-1.5 overflow-hidden">
                                    <div className="w-2 h-2 rounded-full bg-emerald-500 shrink-0"/>
                                    <div className="flex items-baseline gap-1 whitespace-nowrap">
                                        <RollingNumber value={workingDaysCount} />
                                        <span className="text-[10px] text-gray-400 font-bold uppercase tracking-tighter">dni</span>
                                    </div>
                                </div>
                            </div>
                            <div className="bg-gray-50 p-4 rounded-3xl min-w-0">
                                <p className="text-[10px] font-bold text-gray-400 uppercase mb-1">Wolne</p>
                                <div className="text-sm font-black text-rose-500 flex items-center gap-1.5 overflow-hidden">
                                    <div className="w-2 h-2 rounded-full bg-rose-500 shrink-0"/>
                                    <div className="flex items-baseline gap-1 whitespace-nowrap">
                                        <RollingNumber value={busyDays.length} />
                                        <span className="text-[10px] text-gray-400 font-bold uppercase tracking-tighter">dni</span>
                                    </div>
                                </div>
                            </div>
                        </div>

                        <button
                            onClick={() => setIsMobileFullscreen(true)}
                            className="w-full bg-gray-900 text-white py-4 rounded-2xl font-bold flex items-center justify-center gap-2 active:scale-95 transition-all shadow-xl shadow-gray-200"
                        >
                            Edytuj grafik pracy
                        </button>
                    </div>
                </motion.div>
            </div>

            {/* --- FULL SCREEN MODAL --- */}
            <AnimatePresence>
                {isMobileFullscreen && (
                    <motion.div
                        initial={{ y: '100%' }} animate={{ y: 0 }} exit={{ y: '100%' }}
                        onAnimationStart={() => setIsAnimating(true)}
                        onAnimationComplete={() => setIsAnimating(false)}
                        transition={{ type: 'spring', damping: 32, stiffness: 320 }}
                        className="fixed inset-0 bg-white z-[999] md:hidden flex flex-col"
                    >
                        <div className="w-full flex flex-col items-center pt-3 pb-6 border-b bg-white sticky top-0 z-20 shadow-sm">
                            <div className="w-12 h-1.5 bg-gray-200 rounded-full mb-6" />
                            <div className="w-full px-6 flex items-center justify-between">
                                <button onClick={() => setIsMobileFullscreen(false)} className="w-10 h-10 bg-gray-100 rounded-full flex items-center justify-center text-gray-500 active:scale-90"><X size={20} /></button>
                                <h3 className="font-black text-xl tracking-tight text-gray-900">Mój Grafik</h3>
                                <div className="w-10 h-10 bg-indigo-50 rounded-full flex items-center justify-center text-indigo-600"><Info size={18} /></div>
                            </div>
                        </div>

                        <div className="flex-1 overflow-y-auto bg-white pt-2">
                            <RenderCalendarGrid isMobile={true} />

                            <div className="px-8 mt-10 mb-24 grid grid-cols-2 gap-4">
                                <div className="flex items-center gap-2 text-[10px] font-bold text-gray-400 uppercase border border-gray-100 p-3 rounded-2xl">
                                    <div className="w-2 h-2 rounded-full bg-emerald-500" /> Praca
                                </div>
                                <div className="flex items-center gap-2 text-[10px] font-bold text-gray-400 uppercase border border-gray-100 p-3 rounded-2xl">
                                    <div className="w-2 h-2 rounded-full bg-rose-500" /> Wolne
                                </div>
                            </div>
                        </div>

                        <div className="p-6 bg-white border-t flex items-center gap-3 sticky bottom-0 z-30 shadow-[0_-10px_40px_rgba(0,0,0,0.05)]">
                            <button onClick={toggleWeekends} className={`flex-[1.1] py-4 rounded-2xl font-black text-[10px] uppercase tracking-widest transition-all active:scale-95 flex items-center justify-center gap-2 border ${weekendsBlocked ? 'bg-rose-50 text-rose-600 border-rose-100' : 'bg-gray-50 text-gray-500 border-gray-100'}`}>
                                <CalendarDays size={16} /> Weekendy
                            </button>
                            <button onClick={handleSave} disabled={isSaving || isLoading} className="flex-[2] py-4 bg-indigo-600 text-white rounded-2xl font-black text-[10px] uppercase tracking-widest shadow-xl shadow-indigo-100 active:scale-95 transition-all flex items-center justify-center gap-2">
                                {isSaving ? <Loader2 size={16} className="animate-spin" /> : <Save size={16} />} Zapisz grafik
                            </button>
                        </div>
                    </motion.div>
                )}
            </AnimatePresence>
        </>
    );
});
